cmake_minimum_required(VERSION 3.8)
project(ai_robot_base)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(rclcpp REQUIRED)
add_executable(base_status_node src/base_status_node.cpp)
target_include_directories(base_status_node PRIVATE include)
ament_target_dependencies(base_status_node rclcpp)
add_executable(cmd_vel_safety_node src/cmd_vel_safety_node.cpp)
target_include_directories(cmd_vel_safety_node PRIVATE include)
ament_target_dependencies(cmd_vel_safety_node diagnostic_msgs geometry_msgs rclcpp)
add_executable(odom_contract_relay src/odom_contract_relay.cpp)
ament_target_dependencies(odom_contract_relay nav_msgs rclcpp)
install(TARGETS base_status_node cmd_vel_safety_node odom_contract_relay DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(command_watchdog_test test/command_watchdog_test.cpp)
  target_include_directories(command_watchdog_test PRIVATE include)
  add_test(NAME command_watchdog_test COMMAND command_watchdog_test)
  add_executable(command_filter_test test/command_filter_test.cpp)
  target_include_directories(command_filter_test PRIVATE include)
  add_test(NAME command_filter_test COMMAND command_filter_test)
endif()
ament_package()
